\section{The question}
\begin{frame}{Start with the question}
How can repeated measurements improve an estimate?
\begin{itemize}
\item Explain why the problem matters.
\item State the assumptions before presenting the method.
\item Separate predictions from observed evidence.
\end{itemize}
\end{frame}
\section{The method}
\begin{frame}{A simple estimator}
For independent observations $y_i=\theta+\varepsilon_i$,
\[
\widehat\theta=\frac{1}{n}\sum_{i=1}^n y_i,
\qquad \operatorname{Var}(\widehat\theta)=\frac{\sigma^2}{n}.
\]
\begin{block}{Interpretation}
More observations reduce uncertainty when the noise assumptions hold.
\end{block}
\end{frame}
